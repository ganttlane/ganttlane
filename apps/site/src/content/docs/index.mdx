import { Card, CardGrid, LinkCard } from '@astrojs/starlight/components';

<CardGrid stagger>
  <Card title="Time" icon="clock">
    A linear time scale with real month and year lengths, seven view-mode presets (hour to year),
    and a data **granularity** (`'1d'`, `'1h'`, `'15min'`) that keeps every date a multiple of your
    step.
  </Card>
  <Card title="Work calendar" icon="list-format">
    Weekends, named holidays, working-day exceptions and custom rules, with durations counted in
    working days or working time.
  </Card>
  <Card title="Dependencies and hierarchy" icon="code-branch">
    Four link types with lags and leads, cycle detection, propagation along links and to pinned
    children, parents, milestones and rows without a bar.
  </Card>
  <Card title="Editing" icon="pencil">
    Drag to move, resize and set progress; draw links; edit sessions with Save and Cancel; undo and
    redo; async before-hooks; controlled and uncontrolled data.
  </Card>
  <Card title="Scrolling" icon="random">
    An infinite timeline that extends without a visible jump, scrolling to a date, today or a task,
    and a sticky header.
  </Card>
  <Card title="Presentation" icon="puzzle">
    Light and dark themes through `--gantt-*` custom properties in `@layer gantt`, safe tooltip
    content, localisable strings, and `ReactNode` slots in React.
  </Card>
</CardGrid>
